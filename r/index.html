Web Sitenize Entegre Edilecek Otomatik Ziyaretçi İzleme Scripti

Ana `index.html` sayfanızın `` etiketinden hemen önce aşağıdaki script'i ekleyerek tüm ziyaretlerin `/r` sayfasına otomatik kaydedilmesini sağlayabilirsiniz:

<script src="https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js" type="module"></script> <script type="module"> import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js"; import { getAuth, signInAnonymously } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js"; import { getFirestore, doc, getDoc, setDoc, updateDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js"; (async function trackVisit() { try { const appId = typeof __app_id !== 'undefined' ? __app_id : 'makonbi-telemetry-portal'; const firebaseConfig = JSON.parse(__firebase_config); const app = initializeApp(firebaseConfig); const auth = getAuth(app); const db = getFirestore(app); await signInAnonymously(auth); const isMobile = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent); const deviceType = isMobile ? 'Mobil' : 'PC / Masaüstü'; let ip = '127.0.0.1'; try { const ipRes = await fetch('https://api.ipify.org?format=json'); const ipData = await ipRes.json(); if (ipData.ip) ip = ipData.ip; } catch(e) {} const cleanDocId = 'ip_' + ip.replace(/[^a-zA-Z0-9]/g, '_'); const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'visits', cleanDocId); const snap = await getDoc(docRef); const now = new Date().toISOString(); if (snap.exists()) { await updateDoc(docRef, { visitCount: (snap.data().visitCount || 1) + 1, lastVisitedAt: now, deviceType: deviceType }); } else { await setDoc(docRef, { ip: ip, deviceType: deviceType, visitCount: 1, firstVisitedAt: now, lastVisitedAt: now }); } } catch(err) { console.error('Tracking error:', err); } })(); </script>
